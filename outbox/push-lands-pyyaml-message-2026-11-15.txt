---- check.sh
#!/usr/bin/env bash
# The check for "a guard without PyYAML says what to install and stops": with EVERY site-packages disabled (python3 -S),
# the guard must exit 1 AND its own output must carry the remedy. Run from the repository root.
#   bash check.sh [path-to-guard]      default scripts/check-push-lands.py
G=${1:-scripts/check-push-lands.py}
out=$(python3 -S "$G" 2>&1 >/dev/null); e=$?
printf 'guard: %s\nexit: %s\nlast stderr lines:\n' "$G" "$e"
printf '%s\n' "$out" | tail -3 | sed 's/^/  | /'
if [ "$e" = 1 ] && printf '%s' "$out" | grep -qF 'needs PyYAML (python3 -m pip install pyyaml)'; then echo "CHECK PASS (exit 1 and the remedy is named)"; else echo "CHECK FAIL"; fi

---- mut.py
#!/usr/bin/env python3
"""Rule 90 for the missing-PyYAML handling in scripts/check-push-lands.py. Run from the repository root:  python3 mut.py <path to check.sh>
Each mutant is made on a COPY in a temporary directory; the tracked file is only read. Each asserts its anchor is unique and that the copy differs
from the source BEFORE a verdict is printed, and the unmutated file must pass the check first (a positive control), so NOT CAUGHT cannot mean "nothing
was mutated"."""
import pathlib, subprocess, sys, tempfile
CHECK = sys.argv[1]
SRC = pathlib.Path('scripts/check-push-lands.py').read_text()
MSG = 'sys.exit("check-push-lands.py needs PyYAML (python3 -m pip install pyyaml)")'
HANDLER = ('try:\n    import yaml\nexcept ImportError:  # pragma: no cover - the runner image has it; a dev box without it must say so, not skip\n    ' + MSG + '\n')
MUTANTS = [
    ('M1 the remedy is gone from the message', MSG, 'sys.exit("check-push-lands.py needs something")'),
    ('M2 it says so and exits 0, a skip that reads as a pass', MSG, 'print("check-push-lands.py needs PyYAML (python3 -m pip install pyyaml)", file=sys.stderr); sys.exit(0)'),
    ('M3 the handler is deleted, a bare import again', HANDLER, 'import yaml\n'),
]
def verdict(text):
    with tempfile.TemporaryDirectory(prefix='pyyaml-mut-') as td:
        p = pathlib.Path(td) / 'check-push-lands.py'
        p.write_text(text)
        r = subprocess.run(['bash', CHECK, str(p)], capture_output=True, text=True)
    return r.stdout
ctl = verdict(SRC)
print('positive control (the fixed file, unmutated):', 'CHECK PASS' if 'CHECK PASS' in ctl else 'CHECK FAIL')
if 'CHECK PASS' not in ctl:
    print('STOP: the check is red on the unmutated file, so no mutant can be judged'); sys.exit(1)
caught = 0
for name, find, repl in MUTANTS:
    n = SRC.count(find)
    assert n == 1, f'{name}: its anchor occurs {n} times, not once'
    mutated = SRC.replace(find, repl)
    assert mutated != SRC, f'{name}: the mutation changed nothing'
    out = verdict(mutated)
    ok = 'CHECK FAIL' in out and 'CHECK PASS' not in out
    caught += ok
    print(f'{name}: anchor unique, copy differs by {len(mutated) - len(SRC):+d} chars; {"CAUGHT" if ok else "NOT CAUGHT"}')
print(f'caught {caught} of {len(MUTANTS)} mutants')
sys.exit(0 if caught == len(MUTANTS) else 1)
