==== Z1. The replay script, verbatim ====
// Replays brief-phase-live-check.mjs's card matcher against a saved live-check job log, offline.
//   node replay-matcher.mjs <job log> [--and] [--rows=<label>|<label>...]
// The log holds (a) the slate's cards as the weather step prints them: a game-id line, then `teams  score  [status]`,
// and (b) the brief step's table: one row per testable game, `label  expected  shown`, where a row that matched more
// than one card and passed prints `(N cards, duplicated upstream)`.
// The replay applies the check's own predicate to (a) for every row of (b) and compares its card count with the count
// the check printed. It can fail: with --and (both teams must match) the counts change and it must disagree with the log.
import fs from 'node:fs'

const [file, ...flags] = process.argv.slice(2)
const AND = flags.includes('--and')
const SHOW = (flags.find(f => f.startsWith('--rows=')) ?? '').slice(7).split('|').filter(Boolean)
const strip = l => l.replace(/^\d{4}-\d\d-\d\dT[\d:.]+Z /, '')
const lines = fs.readFileSync(file, 'utf8').split('\n').map(strip)

// (a) cards: `  <GAME_ID>` then `    teams  score  [status]`
const cards = []
for (let i = 0; i < lines.length - 1; i++) {
  const idm = lines[i].match(/^ {2}([A-Za-z0-9]+_\d{4}-\d\d-\d\d_\S+)$/)
  if (!idm) continue
  const tm = lines[i + 1].match(/^ {4}(.+?) {2}(—|\d+–\d+) {2}\[(.+)\]$/)
  if (!tm) continue
  cards.push({ id: idm[1], teams: tm[1], score: tm[2], status: tm[3] })
}

// (b) rows: from `  game ... expected ... shown` to the blank line after the table
const start = lines.findIndex(l => /^ {2}game {30,}expected {2,}shown$/.test(l))
const rows = []
const evidence = new Map()
if (start >= 0) {
  let lastEvidenceId = null
  for (let i = start + 1; i < lines.length; i++) {
    const l = lines[i]
    if (l.trim() === '') break
    const ev = l.match(/^ {6}evidence: game\.id="([^"]+)"/)
    if (ev) { lastEvidenceId = ev[1]; continue }
    if (/^ {16}/.test(l)) continue
    const rm = l.match(/^ {2}(.{40})(\S+) +(.+?)( +<-- )?$/)
    if (!rm) continue
    const label = rm[1].trim()
    const [away, home] = label.split(' @ ')
    const n = (rm[3].match(/\((\d+) cards, duplicated upstream\)/) ?? [])[1]
    rows.push({ label, away, home, expected: rm[2], shown: rm[3], red: !!rm[4], printedN: n ? Number(n) : (rm[4] ? null : 1), gameId: rm[4] ? lastEvidenceId : null })
  }
}

const matchOf = (r) => cards.filter(c => AND
  ? (c.teams.includes(r.away) && c.teams.includes(r.home))
  : (c.teams.includes(r.away) || c.teams.includes(r.home)))

let agree = 0, disagree = 0
const bad = []
for (const r of rows.filter(r => !r.red)) {
  const m = matchOf(r)
  if (m.length === r.printedN) agree++
  else { disagree++; bad.push(`${r.label}: log printed ${r.printedN}, replay found ${m.length}`) }
}
const passing = rows.filter(r => !r.red).length
console.log(`log: ${file.split('/').pop()}  matcher: ${AND ? 'BOTH teams (&&)' : "either team (||), the check's own"}`)
console.log(`cards parsed: ${cards.length}; rows parsed: ${rows.length} (${rows.filter(r => r.red).length} red, ${passing} passing)`)
console.log(`replay vs log, passing rows: checked ${passing} of ${passing}; the replay's card count equals the printed one on ${agree}, differs on ${disagree}`)
for (const b of bad.slice(0, 8)) console.log(`  DIFFERS  ${b}`)

for (const r of rows.filter(r => r.red)) {
  const m = matchOf(r)
  const own = cards.find(c => c.id === r.gameId)
  console.log(`\nRED ROW  ${r.label}   (game id ${r.gameId}; the row's own card: ${own ? `${JSON.stringify(own.teams)} [${own.status}]` : 'NOT FOUND in the card list'})`)
  console.log(`  cards the matcher takes for it: ${m.length}`)
  for (const c of m) console.log(`    ${c.id === r.gameId ? 'OWN    ' : 'PARTNER'} ${c.id}  ${c.teams}  [${c.status}]`)
}

for (const label of SHOW) {
  const r = rows.find(x => x.label === label)
  if (!r) { console.log(`\nROW ${label}: not in this log's table`); continue }
  const m = matchOf(r)
  console.log(`\nROW  ${r.label}   (${r.red ? 'red' : 'passing'}; the check printed ${r.printedN ?? 'no card count'}; the replay takes ${m.length})`)
  for (const c of m) console.log(`    ${c.id}  ${c.teams}  [${c.status}]`)
}
