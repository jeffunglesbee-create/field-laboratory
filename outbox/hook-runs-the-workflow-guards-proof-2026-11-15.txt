---- hook-proof.sh
#!/usr/bin/env bash
# Runs the pre-commit hook in a SCRATCH worktree of the branch tip, on a clean tree and on two staged mutations, and prints what the
# hook did. Phase `before` runs the hook as committed; phase `after` applies change.patch first. Run from the repository root.
#   bash hook-proof.sh before|after
set -u
H=/tmp/claude-0/-home-claude/17dcf9de-532b-5e13-b576-6123f7d6b834/scratchpad/hook-e2e
PHASE=${1:?before|after}
WT=$H/wt
REPO=$PWD
OUTF=$H/hook.out
if [ "$PHASE" = before ]; then
  git worktree remove --force "$WT" >/dev/null 2>&1; rm -rf "$WT"
  git worktree add --detach "$WT" HEAD >/dev/null 2>&1 || { echo "REFUSED: no worktree"; exit 2; }
  ln -s "$REPO/node_modules" "$WT/node_modules"
fi
cd "$WT" || exit 2
echo "worktree $(git rev-parse --short HEAD), phase $PHASE; hook has $(grep -c '^npm run' .githooks/pre-commit) npm run lines"
if [ "$PHASE" = after ]; then
  git checkout -q -- .githooks/pre-commit package.json   # back to the committed state, then apply
  git apply "$H/change.patch" || { echo "REFUSED: patch did not apply"; exit 2; }
  echo "change.patch applied; hook now has $(grep -c '^npm run' .githooks/pre-commit) npm run lines; changed files: $(git status --porcelain | awk '{print $2}' | tr '\n' ' ')"
fi
BASE_CHANGED=$(git status --porcelain | awk '{print $2}' | sort | tr '\n' ' ')

restore() {
  git reset -q -- scripts/check-script-orphans.mjs .github/workflows/bdb-census.yml outbox/hook-proof-note.txt 2>/dev/null
  git checkout -q -- scripts/check-script-orphans.mjs .github/workflows/bdb-census.yml
  rm -f outbox/hook-proof-note.txt
  NOW=$(git status --porcelain | awk '{print $2}' | sort | tr '\n' ' ')
  [ "$NOW" = "$BASE_CHANGED" ] && echo "restored: changed files are the same as before the mutation (${BASE_CHANGED:-none})" || { echo "RESTORE FAILED: now [$NOW], expected [$BASE_CHANGED]"; exit 2; }
}
mut_script() {
  local f=scripts/check-script-orphans.mjs
  [ "$(grep -c '__mutant' $f)" = 0 ] || { echo "REFUSED: already mutated"; exit 2; }
  printf 'import { chromium } from "playwright"\nconst __mutant = () => chromium.launch()\n' | cat - $f > $f.new && mv $f.new $f
  [ "$(grep -c '__mutant' $f)" = 1 ] || { echo "REFUSED: mutation not applied"; exit 2; }
  echo "hook-proof: the outbox note a source commit carries" > outbox/hook-proof-note.txt
  git add $f outbox/hook-proof-note.txt
  echo "MUTATION script-only: 2 lines added at the top of $f (an ungated verify step's entry script); staged: $(git diff --cached --name-only | sort | tr '\n' ' ')"
}
mut_workflow() {
  local f=.github/workflows/bdb-census.yml
  python3 - "$f" <<'PY' || exit 2
import sys
p = sys.argv[1]
s = open(p).read()
a = 'git merge-base --is-ancestor HEAD "origin/$BRANCH"'
if s.count(a) != 1:
    print(f"REFUSED: anchor occurs {s.count(a)} times, not 1"); sys.exit(2)
m = s.replace(a, 'git merge-base HEAD "origin/$BRANCH"')
if m == s:
    print("REFUSED: nothing changed"); sys.exit(2)
open(p, 'w').write(m)
print("applied: the landing verification after the push-retry loop in", p, "no longer matches (merge-base --is-ancestor -> merge-base)")
PY
  echo "hook-proof: the outbox note a source commit carries" > outbox/hook-proof-note.txt
  git add $f outbox/hook-proof-note.txt
  echo "MUTATION workflow: staged: $(git diff --cached --name-only | sort | tr '\n' ' ')"
}
run_hook() {
  local t0 t1 e
  t0=$(date +%s.%N)
  bash .githooks/pre-commit > "$OUTF" 2>&1; e=$?
  t1=$(date +%s.%N)
  echo "  what the hook's output holds, as line counts: browser guard passed [verdict: gated] $(grep -c '^verdict: gated' "$OUTF"); browser guard failed [needs a browser and must carry] $(grep -c 'needs a browser and must carry' "$OUTF"); push-lands passed [loops verify that the push landed] $(grep -c 'push-retry loop(s) verify that the push landed' "$OUTF"); push-lands failed [nothing after them] $(grep -c 'with nothing after them' "$OUTF")"
  echo "  hook output, last 12 lines:"; tail -12 "$OUTF" | cut -c1-210 | sed 's/^/    /'
  awk -v a="$t0" -v b="$t1" -v e="$e" 'BEGIN{printf "  hook exit: %s; elapsed %.1f s\n", e, b-a}'
  LAST_EXIT=$e
}
guards_direct() {
  python3 scripts/check-browser-gate.py > /dev/null 2>&1; echo "  the same tree, check-browser-gate.py run directly: exit $?"
  python3 scripts/check-push-lands.py > /dev/null 2>&1; echo "  the same tree, check-push-lands.py run directly: exit $?"
}
