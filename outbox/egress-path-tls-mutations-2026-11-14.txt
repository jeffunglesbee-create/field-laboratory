      const p = k >= 0 ? src[k] : ';'
      const w = /[A-Za-z_$]/.test(p) ? prevWord(k) : ''
      if (REGEX_PREV.has(p) || ['return', 'typeof', 'case', 'in', 'of'].includes(w)) {
        // regex literal: to the next unescaped slash outside a class
        let j = i + 1, cls = false
        while (j < src.length && src[j] !== '\n') { if (src[j] === '\\') { j += 2; continue } if (src[j] === '[') cls = true; else if (src[j] === ']') cls = false; else if (src[j] === '/' && !cls) break; j++ }
        i = j + 1; while (/[a-z]/.test(src[i] ?? '')) i++
        continue
      }
    }
    isCode[i] = 1; i++; continue
  }
  if (mode === 'line') { if (c === '\n') mode = 'code'; i++; continue }
  if (mode === 'block') { if (c === '*' && n === '/') { mode = 'code'; i += 2; continue } i++; continue }
  if (mode === 'sq' || mode === 'dq') { if (c === '\\') { i += 2; continue } if ((mode === 'sq' && c === "'") || (mode === 'dq' && c === '"') || c === '\n') mode = 'code'; i++; continue }
  if (mode === 'tpl') {
    if (c === '\\') { i += 2; continue }
    if (c === '`') { mode = modes.pop() ?? 'code'; i++; continue }
    if (c === '$' && n === '{') { braceDepth.push(0); mode = 'code'; i += 2; continue }
    i++; continue
  }
}
// the ${ } bookkeeping above pushes a marker per expression; they are popped implicitly by the closing brace, so clean the stack of markers
// line starts, to map offsets to line numbers
const lineStarts = [0]; for (let k = 0; k < src.length; k++) if (src[k] === '\n') lineStarts.push(k + 1)
const lineOf = off => { let lo = 0, hi = lineStarts.length - 1; while (lo < hi) { const mid = (lo + hi + 1) >> 1; if (lineStarts[mid] <= off) lo = mid; else hi = mid - 1 } return lo + 1 }
const selfTestStart = src.split('\n').findIndex(l => l.startsWith("if (isMain(import.meta.url) && process.argv.includes('--self-test'))")) + 1
const OPS = [
  ['&&', ['||']], ['||', ['&&']],
  ['===', ['!==']], ['!==', ['===']],
  ['>=', ['>', '<']], ['<=', ['<', '>']],
]
const mutants = []
const push = (off, from, to, label) => {
  const ln = lineOf(off)
  if (!added.has(ln) || ln >= selfTestStart) return
  mutants.push({ off, ln, from, to, label })
}
for (let k = 0; k < src.length; k++) {
  if (!isCode[k]) continue
  const three = src.slice(k, k + 3), two = src.slice(k, k + 2), one = src[k]
  if (three === '===' || three === '!==') { if (isCode[k + 1] && isCode[k + 2]) push(k, three, three === '===' ? '!==' : '===', 'eq'); k += 2; continue }
  if (two === '&&' || two === '||') { push(k, two, two === '&&' ? '||' : '&&', 'logic'); k += 1; continue }
  if (two === '>=' || two === '<=') { for (const to of two === '>=' ? ['>', '<'] : ['<', '>']) push(k, two, to, 'cmp'); k += 1; continue }
  if (two === '=>' || two === '==' || two === '!=' || two === '??' || two === '++' || two === '--' || two === '<<' || two === '>>') { k += 1; continue }
  if (one === '>' || one === '<') { const p = src[k - 1], nx = src[k + 1]; if (p === '=' || p === '-' || nx === '=') continue; for (const to of one === '>' ? ['>=', '<='] : ['<=', '>=']) push(k, one, to, 'cmp'); continue }
  if (one === '!' && src[k + 1] !== '=' && src[k + 1] !== undefined) { push(k, '!', '', 'not'); continue }
  const tx = src.slice(k, k + 10)
  const sw = /^(Math\.min\(|Math\.max\(|\.some\(|\.every\()/.exec(tx)
  if (sw && (isCode[k + 1] || sw[1].length < 2)) { const swap = { 'Math.min(': 'Math.max(', 'Math.max(': 'Math.min(', '.some(': '.every(', '.every(': '.some(' }[sw[1]]; push(k, sw[1], swap, 'call'); k += sw[1].length - 1; continue }
  const m = /^(true|false)\b/.exec(src.slice(k, k + 6))
  if (m && !/[\w$.]/.test(src[k - 1] ?? ' ')) { push(k, m[1], m[1] === 'true' ? 'false' : 'true', 'bool'); k += m[1].length - 1; continue }
}
fs.writeFileSync(outFile, JSON.stringify({ selfTestStart, codeChars: isCode.reduce((a, b) => a + b, 0), mutants }, null, 0))
const by = {}; for (const m of mutants) by[m.label] = (by[m.label] ?? 0) + 1
console.log(`added library lines considered: ${[...added].filter(l => l < selfTestStart).length}; mutants: ${mutants.length} ${JSON.stringify(by)}; self-test starts at line ${selfTestStart}`)

// ---- sweep-run.mjs ----
// Run every generic mutant of sweep-gen.mjs against the library's own self-test, in its own copy, two at a time.
//   node sweep-run.mjs <lib file> <mutants.json> <out json>
import fs from 'node:fs'
import os from 'node:os'
import path from 'node:path'
import { spawn } from 'node:child_process'
const [libFile, mutFile, outFile] = process.argv.slice(2)
const root = path.dirname(libFile)
const src = fs.readFileSync(libFile, 'utf8')
const { mutants } = JSON.parse(fs.readFileSync(mutFile, 'utf8'))
const lines = src.split('\n')
const WORKERS = 2
const dirs = Array.from({ length: WORKERS }, () => { const d = fs.mkdtempSync(path.join(os.tmpdir(), 'sweep-')); for (const f of ['checks.mjs', 'render-check.mjs']) fs.copyFileSync(path.join(root, f), path.join(d, f)); return d })
const runIn = (dir, text) => new Promise(resolve => {
  fs.writeFileSync(path.join(dir, 'egress-path-lib.mjs'), text)
  const p = spawn(process.execPath, [path.join(dir, 'egress-path-lib.mjs'), '--self-test'], { stdio: ['ignore', 'pipe', 'pipe'] })
  let out = ''; p.stdout.on('data', d => out += d); p.stderr.on('data', d => out += d)
  const timer = setTimeout(() => p.kill('SIGKILL'), 90000)
  p.on('close', (code, sig) => { clearTimeout(timer); resolve({ code, sig, failing: out.split('\n').filter(l => l.startsWith('FAIL  ')).map(l => l.slice(6)), tail: out.trim().split('\n').slice(-1)[0] }) })
})
const base = await runIn(dirs[0], src)
if (base.code !== 0) { console.log(`the unmutated copy does not pass (${base.tail}); nothing was run`); process.exit(2) }
console.log(`unmutated copy: exit 0, "${base.tail}"`)
const results = new Array(mutants.length)
let next = 0
await Promise.all(dirs.map(async dir => {
  for (;;) {
    const i = next++; if (i >= mutants.length) return
    const m = mutants[i]
    if (src.slice(m.off, m.off + m.from.length) !== m.from) { results[i] = { ...m, status: 'ANCHOR-MOVED' }; continue }
    const text = src.slice(0, m.off) + m.to + src.slice(m.off + m.from.length)
    if (text === src) { results[i] = { ...m, status: 'NOT-APPLIED' }; continue }
    const r = await runIn(dir, text)
    results[i] = { ...m, status: r.code === 0 ? 'survived' : r.failing.length ? 'killed' : 'crashed', failing: r.failing.length, first: r.failing[0] ?? r.tail, sig: r.sig }
  }
}))
for (const d of dirs) fs.rmSync(d, { recursive: true, force: true })
const by = {}; for (const r of results) by[r.status] = (by[r.status] ?? 0) + 1
fs.writeFileSync(outFile, JSON.stringify(results.map(r => ({ ...r, text: lines[r.ln - 1].trim().slice(0, 200) }))))
console.log(`checked ${results.length} of ${mutants.length} generic mutants; ${JSON.stringify(by)}`)
for (const r of results.filter(r => r.status !== 'killed')) console.log(`${r.status.toUpperCase().padEnd(9)} line ${r.ln} [${r.label}] ${r.from} -> ${r.to || '(removed)'} : ${lines[r.ln - 1].trim().slice(0, 140)}`)
