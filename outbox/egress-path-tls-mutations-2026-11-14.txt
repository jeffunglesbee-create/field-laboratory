// An application of every entry that does not use the harness: a copy of the library in a temp directory, the replacement made by this script (anchor asserted unique and applied), the self-test run on the copy.
import fs from 'node:fs'
import os from 'node:os'
import path from 'node:path'
import { spawnSync } from 'node:child_process'
import { ENTRIES } from './muts-tls.mjs'
const root = '/home/claude/field-laboratory/scripts'
const lib = fs.readFileSync(`${root}/egress-path-lib.mjs`, 'utf8')
const dir = fs.mkdtempSync(path.join(os.tmpdir(), 'muts-by-hand-'))
for (const f of ['checks.mjs', 'render-check.mjs']) fs.copyFileSync(`${root}/${f}`, `${dir}/${f}`)
const run = src => {
  fs.writeFileSync(`${dir}/egress-path-lib.mjs`, src)
  const r = spawnSync(process.execPath, [`${dir}/egress-path-lib.mjs`, '--self-test'], { encoding: 'utf8', timeout: 60000 })
  const out = `${r.stdout}${r.stderr}`
  return { status: r.status, failing: out.split('\n').filter(l => l.startsWith('FAIL  ')).map(l => l.slice(6)), crashed: /Error|Exception/.test(r.stderr) && !/^FAIL/m.test(out), tail: out.trim().split('\n').slice(-1)[0] }
}
const clean = run(lib)
console.log(`unmutated copy: exit ${clean.status}, ${clean.failing.length} failing, "${clean.tail}"`)
let caught = 0
for (const [id, , find, replace] of ENTRIES) {
  const n = lib.split(find).length - 1
  if (n !== 1) { console.log(`ANCHOR ${n}: ${id}`); continue }
  const mutated = lib.replace(find, () => replace)
  if (mutated === lib) { console.log(`NOT APPLIED: ${id}`); continue }
  const r = run(mutated)
  const red = r.status !== 0
  if (red) caught++
  console.log(`${red ? 'RED ' : 'NOT CAUGHT'} egress-path-${id}: exit ${r.status}, ${r.failing.length} check(s) failing${r.crashed ? ' (a crash, not a check)' : ''}; first: ${(r.failing[0] ?? r.tail).slice(0, 110)}`)
}
console.log(`caught ${caught} of ${ENTRIES.length} when applied by this script to a copy`)
fs.rmSync(dir, { recursive: true, force: true })
