---- harness: sweep.sh, verbatim
#!/usr/bin/env bash
# Reads the brief step of the last 20 SCHEDULE-event runs of live-check over REST (the sandbox cannot reach the deployed page), one row per run.
# Run from the repository root:  bash sweep.sh <work dir>      prints the table; saves each job log in <work dir>.
D=${1:?work dir}
mkdir -p "$D"
gh run list --workflow live-check.yml --event schedule --limit 20 --json databaseId,createdAt,conclusion --jq '.[] | "\(.databaseId) \(.createdAt) \(.conclusion)"' > "$D/runs.txt"
while read -r id created concl; do
  jid=$(gh run view "$id" --json jobs --jq '[.jobs[] | select(.name=="live")][0].databaseId' 2>/dev/null)
  if [ -z "$jid" ] || [ "$jid" = "null" ]; then echo "$id $created $concl NO-JOB"; continue; fi
  if ! gh api "repos/jeffunglesbee-create/field-laboratory/actions/jobs/$jid/logs" > "$D/$id.log" 2>/dev/null; then echo "$id $created $concl LOG-UNREADABLE"; continue; fi
  L=$(sed -E 's/^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9:.]+Z //' "$D/$id.log")
  slate=$(grep -oE 'slate 20[0-9-]+ — [0-9]+ game\(s\)' <<<"$L" | head -1 | sed -E 's/slate (20[0-9-]+) — ([0-9]+) game.*/\1 games=\2/')
  both=$(grep -oE 'final games carrying BOTH a during-play and an after-play brief: [0-9]+' <<<"$L" | head -1 | grep -oE '[0-9]+$')
  if grep -q 'brief-phase rule NOT OBSERVABLE' <<<"$L"; then v=NOT-OBSERVABLE
  elif grep -qE 'FAIL.*every final game with both phases' <<<"$L"; then v=FAIL
  elif grep -qE 'PASS.*every final game with both phases' <<<"$L"; then v=PASS
  else v=NO-BRIEF-LINES; fi
  echo "$id $created $concl slate=${slate:-none} testable=${both:-none} brief=$v"
done < "$D/runs.txt"
