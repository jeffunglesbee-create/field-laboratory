// Serves a page and a relay for scripts/brief-phase-live-check.mjs, so its executable half runs end to end
// (browser, DOM read, relay fetch, verdict, print) against a fixture reconstructed from the 2026-11-14 00:32Z
// live-check render (run 43658647710). Ids, team lines and statuses are from that run's log. Two kinds are NOT in the
// log and are deduced (marked below); everything else the page and relay carry is built here, not observed.
//
//   node serve.mjs <port> [variant]
//   variants:  ok          every card carries its game's id
//              idmismatch  the `California @ Virginia` card carries another id than its game's
import http from 'node:http'

const [port, variant = 'ok'] = process.argv.slice(2)
const S = 'Start time passed, no score served'
const cards = [
  { id: 'CFB_2026-11-14_e401856889', teams: 'Alabama @ Vanderbilt',     status: 'Final', kind: 'game_recap' },
  { id: 'CFB_2026-11-14_e401871100', teams: 'Troy @ South Alabama',     status: S,       kind: 'game_live' },
  { id: 'CFB_2026-11-14_e401858523', teams: 'Michigan @ Oregon',        status: 'Final', kind: 'game_recap' },
  { id: 'CFB_2026-11-14_e401858521', teams: 'Washington @ Michigan St', status: S,       kind: 'game_live' },
  { id: 'CFB_2026-11-14_e401860914', teams: 'Oregon St @ Boise St',     status: S,       kind: 'game_live' },
  { id: 'CFB_2026-11-14_e401858295', teams: 'California @ Virginia',    status: 'Final', kind: 'game_recap' },
  { id: 'CFB_2026-11-14_e401856861', teams: 'Kansas @ West Virginia',   status: 'Final', kind: 'game_recap' },   // kind deduced
  { id: 'CFB_2026-11-14_e401858299', teams: 'Stanford @ Virginia Tech', status: S,       kind: 'game_live' },    // kind deduced
]
const testable = new Set(['Alabama @ Vanderbilt', 'Michigan @ Oregon', 'California @ Virginia'])

const shownId = c => (variant === 'idmismatch' && c.teams === 'California @ Virginia') ? 'CFB_2026-11-14_WRONG' : c.id
const page = `<!doctype html><html><head><meta charset="utf-8"><link rel="icon" href="data:,"><title>fixture</title></head><body>
<span id="viewed-date">2026-11-14</span>
${cards.map(c => `<div class="desk-card" data-game-id="${shownId(c)}"><div class="desk-card__teams">${c.teams}</div><div class="desk-card__status">${c.status}</div>${c.kind ? `<div class="desk-card__brief-kind">${c.kind}</div>` : ''}</div>`).join('\n')}
</body></html>`

const games = cards.map(c => {
  const [away, home] = c.teams.split(' @ ')
  const final = testable.has(c.teams) || c.teams === 'Kansas @ West Virginia'
  return { id: c.id, away, home, espn_event_id: Number(c.id.split('_e')[1]), finalized_at: final ? '2026-11-14 23:30:00' : null, sport: 'CFB' }
})
const briefs = []
for (const c of cards) {
  const live = { game_id: c.id, brief_type: 'game_live', brief_text: 'written while it was on' }
  const recap = { game_id: c.id, brief_type: 'game_recap', brief_text: 'written after it ended' }
  if (testable.has(c.teams)) briefs.push(live, recap)
  else briefs.push(c.kind === 'game_live' ? live : recap)
}
const payload = JSON.stringify({ games: { regular: games, postseason: [] }, briefs })

http.createServer((req, res) => {
  if (req.url === '/' || req.url.startsWith('/?')) { res.setHeader('content-type', 'text/html'); res.end(page) }
  else if (req.url === '/context/date/2026-11-14') { res.setHeader('content-type', 'application/json'); res.end(payload) }
  else { res.statusCode = 404; res.end('not found') }
}).listen(Number(port), '127.0.0.1')

---- mut.mjs
// Mutation run for the self-test of scripts/brief-phase-live-check.mjs, on a COPY outside the repository.
//   node mut.mjs <spec.mjs>      (run from the repository root; the spec default-exports { file, mutants: [{ name, anchor, with }] })
// Every mutant is refused, and the whole run with it (exit 2), unless its anchor occurs exactly once in the file and the
// mutated text differs from the original. A baseline run of the unmutated copy must exit 0 first. A mutant counts as
// caught when the self-test of the mutated copy exits non-zero. The last line is `caught K of N`.
import fs from 'node:fs'
import path from 'node:path'
import { pathToFileURL } from 'node:url'
import { spawnSync } from 'node:child_process'

const specPath = path.resolve(process.argv[2])
const spec = (await import(pathToFileURL(specPath).href)).default
const src = fs.readFileSync(spec.file, 'utf8')
const root = path.join(path.dirname(specPath), 'mutwt')
fs.rmSync(root, { recursive: true, force: true })
fs.mkdirSync(path.join(root, 'scripts'), { recursive: true })
fs.copyFileSync(path.join(path.dirname(spec.file), 'checks.mjs'), path.join(root, 'scripts/checks.mjs'))
fs.symlinkSync(path.resolve('node_modules'), path.join(root, 'node_modules'))
const target = path.join(root, 'scripts/brief-phase-live-check.mjs')
const run = text => { fs.writeFileSync(target, text); return spawnSync('node', [target, '--self-test'], { encoding: 'utf8' }) }

const base = run(src)
console.log(`baseline, no mutation: exit ${base.status}; ${(base.stdout.match(/\d+\/\d+ checks passed/) ?? ['no summary'])[0]}`)
if (base.status !== 0) { console.log('REFUSED: the unmutated copy does not pass'); process.exit(2) }

let caught = 0
for (const m of spec.mutants) {
  const n = src.split(m.anchor).length - 1
  if (n !== 1) { console.log(`REFUSED: the anchor of "${m.name}" occurs ${n} times, not 1`); process.exit(2) }
  const mutated = src.replace(m.anchor, () => m.with)
  if (mutated === src) { console.log(`REFUSED: "${m.name}" changed nothing`); process.exit(2) }
  const r = run(mutated)
  const red = (r.stdout.match(/^FAIL {2}.*$/gm) ?? []).map(l => l.slice(6))
  const isCaught = r.status !== 0
  if (isCaught) caught++
  console.log(`${isCaught ? 'CAUGHT    ' : 'NOT CAUGHT'}  ${m.name}\n             exit ${r.status}; ${red.length} check(s) red${red.map(f => `\n               - ${f.slice(0, 100)}`).join('')}`)
}
console.log(`\ncaught ${caught} of ${spec.mutants.length}`)
process.exit(caught === spec.mutants.length ? 0 : 1)
